   344→/** Tombol nav level-1 (Beranda / Admin) dengan underline motion. */
   345→function NavButton({
   346→  active,
   347→  gold,
   348→  onClick,
   349→  children,
   350→}: {
   351→  active: boolean
   352→  gold: boolean
   353→  onClick: () => void
   354→  children: React.ReactNode
   355→}) {
   356→  return (
   357→    <button
   358→      onClick={onClick}
   359→      aria-current={active ? 'page' : undefined}
   360→      className={cn(
   361→        'relative flex items-center gap-1.5 rounded-lg px-3 py-2 text-sm font-medium transition-colors',
   362→        active
   363→          ? 'text-foreground'
   364→          : gold
   365→            ? 'text-amber-300/90 hover:text-amber-200'
   366→            : 'text-muted-foreground hover:text-foreground'
   367→      )}
   368→    >
   369→      {children}
   370→      {active && (
   371→        <motion.span
   372→          layoutId="nav-underline"
   373→          className="absolute inset-x-2 -bottom-[13px] h-0.5 rounded-full bg-gradient-to-r from-emerald-400 to-amber-300"
   374→        />
   375→      )}
   376→    </button>
   377→  )
   378→}

   380→/** Trigger + dropdown grup (desktop). Menampilkan semua modul grup
   381→ *  lengkap dengan deskripsi & status LIVE / SOON. */
   382→function GroupDropdown({
   383→  group,
   384→  active,
   385→  onNavigate,
   386→}: {
   387→  group: MenuGroup
   388→  active: boolean
   389→  onNavigate: (v: ViewId) => void
   390→}) {
   391→  const GroupIcon = group.icon
   392→  const readyCount = group.items.filter((i) => i.status === 'ready').length
   393→
   394→  // ID stabil eksplisit pada trigger → menimpa ID acak generate-an Radix
   395→  // (useId) sehingga SSR & client selalu identik — mencegah hydration
   396→  // mismatch. suppressHydrationWarning: pengaman untuk atribut internal
   397→  // Radix (mis. aria-controls) yang bisa beda saat dev/HMR & tak berdampak.
   398→  return (
   399→    <DropdownMenu>
   400→      <DropdownMenuTrigger asChild>
   401→        <button
   402→          id={`nav-${group.id}`}
   403→          suppressHydrationWarning
   404→          aria-haspopup="menu"
   405→          className={cn(
   406→            'relative flex items-center gap-1.5 rounded-lg px-3 py-2 text-sm font-medium transition-colors outline-none',
   407→            active
   408→              ? 'text-foreground'
   409→              : 'text-muted-foreground hover:text-foreground data-[state=open]:text-foreground'
   410→          )}
   411→        >
   412→          <GroupIcon className="h-4 w-4" />
   413→          {group.label}
   414→          <ChevronDown className="h-3.5 w-3.5 opacity-60" />
   415→          {active && (
   416→            <motion.span
   417→              layoutId="nav-underline"
   418→              className="absolute inset-x-2 -bottom-[13px] h-0.5 rounded-full bg-gradient-to-r from-emerald-400 to-amber-300"
   419→            />
   420→          )}
   421→        </button>
   422→      </DropdownMenuTrigger>
   423→
   424→      <DropdownMenuContent
   425→        align="start"
   426→        sideOffset={14}
   427→        className="w-[22rem] border-white/10 bg-background/95 p-2 backdrop-blur-xl"
   428→      >
   429→        {/* header grup */}
   430→        <div className="flex items-center justify-between gap-2 px-2 py-2">
   431→          <div className="flex items-center gap-2.5">
   432→            <span className="text-lg">{group.emoji}</span>
   433→            <div>
   434→              <p className="text-sm leading-tight font-bold">{group.label}</p>
   435→              <p className="text-[11px] leading-tight text-muted-foreground">{group.tagline}</p>
   436→            </div>
   437→          </div>
   438→          <span className="font-mono text-[9px] tracking-[0.2em] text-emerald-300/80 uppercase">
   439→            {readyCount}/{group.items.length} live
   440→          </span>
   441→        </div>
   442→        <DropdownMenuSeparator className="bg-white/5" />
   443→
   444→        {group.items.map((item) => (
   445→          <DropdownItem key={item.id} item={item} onNavigate={onNavigate} />
   446→        ))}
   447→      </DropdownMenuContent>
   448→    </DropdownMenu>
   449→  )
   450→}

   452→/** Baris item di dalam dropdown. 'soon' tetap bisa diklik → ComingSoon. */

   454→  const Icon = item.icon

   462→          soon ? 'focus:bg-white/[0.04]' : 'focus:bg-emerald-400/10',
   463→          item.gold && 'focus:bg-amber-400/10'

   466→      <span
   467→        className={cn(
   468→          'mt-0.5 flex h-9 w-9 shrink-0 items-center justify-center rounded-lg border',
   469→          item.gold
   470→            ? 'border-amber-300/30 bg-amber-400/10 text-amber-300'
   471→            : soon
   472→              ? 'border-white/10 bg-white/5 text-muted-foreground'
   473→              : 'border-emerald-400/30 bg-emerald-400/10 text-emerald-300'
   474→        )}
   475→      >
   476→        <Icon className="h-4 w-4" />
   477→      </span>
   478→      <span className="min-w-0 flex-1">
   479→        <span className="flex items-center gap-2">
   480→          <span className={cn('truncate text-sm font-semibold', soon && 'text-foreground/70')}>{item.label}</span>
   481→          {soon ? (
   482→            <span className="shrink-0 rounded-full border border-amber-300/30 bg-amber-400/10 px-1.5 py-0.5 font-mono text-[8px] font-bold tracking-[0.15em] text-amber-300 uppercase">
   483→              Soon
